# Deployhaven M0 (İskelet) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deployhaven monorepo iskeletini kur; çalışan bir NestJS API + Next.js panel, Postgres şeması ve e-posta/parola oturum kimlik doğrulaması (RBAC temeli) ile uçtan uca test edilebilir bir taban üret.

**Architecture:** pnpm workspaces + Turborepo monorepo. `apps/api` (NestJS) kimlik doğrulama ve iş mantığını, `apps/panel` (Next.js App Router) arayüzü, `packages/db` (Prisma) paylaşılan veri erişimini barındırır. Auth, argon2 ile hash'lenen parola + DB destekli oturum çerezi (HTTP-only) üzerine kuruludur. Kayıtta kullanıcıya varsayılan bir Team + `OWNER` üyeliği açılır (RBAC temeli).

**Tech Stack:** TypeScript (strict), pnpm, Turborepo, NestJS 10, Prisma + PostgreSQL, Redis (M0'da yalnız altyapı olarak ayağa kalkar), argon2, class-validator, Vitest + supertest, Next.js 14, Tailwind, shadcn/ui.

## Global Constraints

- Dil/isimlendirme: paket adları `@deployhaven/*` scope'u altında.
- Tüm TypeScript `strict: true`.
- Node 20+ / pnpm 9+.
- Secrets/parolalar asla düz metin loglanmaz; parola yalnız argon2 hash olarak saklanır.
- Her task kendi test döngüsüyle biter ve ayrı commit alır.
- Reverse proxy Traefik, build motoru Nixpacks (M0'da kullanılmaz, sonraki milestone'lar).

- [ ] **Step 2: `apps/api/src/auth/auth.service.ts` yaz**

```ts
import { ConflictException, Injectable } from "@nestjs/common";
import { prisma, Role } from "@deployhaven/db";
import { hashPassword } from "./password.js";

const SESSION_TTL_MS =
  Number(process.env.SESSION_TTL_HOURS ?? 168) * 60 * 60 * 1000;

@Injectable()
export class AuthService {
  async register(email: string, password: string) {
    const existing = await prisma.user.findUnique({ where: { email } });
    if (existing) throw new ConflictException("email already registered");

    const passwordHash = await hashPassword(password);
    return prisma.$transaction(async (tx) => {
      const user = await tx.user.create({ data: { email, passwordHash } });
      const team = await tx.team.create({ data: { name: `${email}'s team` } });
      await tx.membership.create({
        data: { userId: user.id, teamId: team.id, role: Role.OWNER },
      });
      return { userId: user.id, teamId: team.id };
    });
  }

  async createSession(userId: string) {
    const expiresAt = new Date(Date.now() + SESSION_TTL_MS);
    return prisma.session.create({ data: { userId, expiresAt } });
  }
}
```

- [ ] **Step 3: `apps/api/src/auth/auth.controller.ts` yaz**

```ts
import { Body, Controller, HttpCode, Post, Res } from "@nestjs/common";
import type { Response } from "express";
import { CredentialsDto } from "./dto.js";
import { AuthService } from "./auth.service.js";

const COOKIE = process.env.SESSION_COOKIE_NAME ?? "dh_session";

@Controller("auth")
export class AuthController {
  constructor(private readonly auth: AuthService) {}

  @Post("register")
  @HttpCode(201)
  async register(
    @Body() body: CredentialsDto,
    @Res({ passthrough: true }) res: Response,
  ) {
    const result = await this.auth.register(body.email, body.password);
    const session = await this.auth.createSession(result.userId);
    res.cookie(COOKIE, session.id, {
      httpOnly: true,
      sameSite: "lax",
      expires: session.expiresAt,
    });
    return result;
  }
}
```

- [ ] **Step 4: `apps/api/src/auth/session.guard.ts` yaz**

```ts
import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from "@nestjs/common";
import type { Request } from "express";
import { AuthService } from "./auth.service.js";

const COOKIE = process.env.SESSION_COOKIE_NAME ?? "dh_session";

@Injectable()
export class SessionGuard implements CanActivate {
  constructor(private readonly auth: AuthService) {}

  async canActivate(ctx: ExecutionContext): Promise<boolean> {
    const req = ctx.switchToHttp().getRequest<Request & { userId?: string }>();
    const sessionId = req.cookies?.[COOKIE];
    if (!sessionId) throw new UnauthorizedException();
    const user = await this.auth.getUserFromSession(sessionId);
    if (!user) throw new UnauthorizedException();
    req.userId = user.id;
    return true;
  }
}
```

```tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { apiFetch } from "@/lib/api";

export default function LoginPage() {
  const router = useRouter();
  const [mode, setMode] = useState<"login" | "register">("login");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    const res = await apiFetch(`/auth/${mode}`, {
      method: "POST",
      body: JSON.stringify({ email, password }),
    });
    if (res.ok) router.push("/");
    else setError(mode === "login" ? "Giriş başarısız" : "Kayıt başarısız");
  }

  return (
    <main className="mx-auto mt-24 max-w-sm p-6">
      <h1 className="mb-4 text-xl font-semibold">
        {mode === "login" ? "Giriş" : "Kayıt"}
      </h1>
      <form onSubmit={submit} className="space-y-3">
        <input
          className="w-full rounded border p-2"
          type="email"
          placeholder="E-posta"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
          required
        />
        <input
          className="w-full rounded border p-2"
          type="password"
          placeholder="Parola"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          required
        />
        {error && <p className="text-sm text-red-600">{error}</p>}
        <button className="w-full rounded bg-black p-2 text-white" type="submit">
          {mode === "login" ? "Giriş yap" : "Kayıt ol"}
        </button>
      </form>
      <button
        className="mt-3 text-sm text-gray-600 underline"
        onClick={() => setMode(mode === "login" ? "register" : "login")}
      >
        {mode === "login" ? "Hesabın yok mu? Kayıt ol" : "Zaten üye misin? Giriş yap"}
      </button>
    </main>
  );
}
```
